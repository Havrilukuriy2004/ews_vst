' Завантаження списку груп із файлу
Private Sub LoadGroups()
    Dim sourceWorkbook As Workbook
    Dim sourceSheet As Worksheet
    Dim lastRow As Long
    Dim lastCol As Long
    Dim i As Long, j As Long
    Dim groups As Object
    Dim groupName As Variant
    Dim headerRow As Long
    
    ' Шлях до файлу на робочому столі
    Dim filePath As String
    filePath = Environ("USERPROFILE") & "\Desktop\counterpartGroups.xlsx"
    
    ' Оновлюємо статус
    InfoLabel.Caption = "Завантаження файлу counterpartGroups.xlsx..."
    InfoLabel.ForeColor = RGB(0, 0, 255)  ' Синій колір
    DoEvents
    
    ' Перевіряємо чи існує файл
    If Dir(filePath) = "" Then
        InfoLabel.Caption = "ПОМИЛКА: Файл не знайдено: " & filePath
        InfoLabel.ForeColor = RGB(255, 0, 0)  ' Червоний колір
        Exit Sub
    End If
    
    ' Відкриваємо файл-джерело
    On Error GoTo ErrorHandler
    Set sourceWorkbook = Workbooks.Open(filePath, ReadOnly:=True)
    Set sourceSheet = sourceWorkbook.Worksheets(1)
    
    ' Визначаємо останній заповнений рядок і стовпець
    lastRow = sourceSheet.Cells(sourceSheet.Rows.Count, 1).End(xlUp).Row
    lastCol = sourceSheet.Cells(1, sourceSheet.Columns.Count).End(xlToLeft).Column
    
    ' Шукаємо заголовки стовпців (у першому рядку)
    headerRow = 1
    groupColumn = 0
    codeColumn = 0
    
    For j = 1 To lastCol
        If sourceSheet.Cells(headerRow, j).Value = "Група" Then
            groupColumn = j
        ElseIf sourceSheet.Cells(headerRow, j).Value = "ЄДРПОУ" Then
            codeColumn = j
        End If
    Next j
    
    ' Перевіряємо, чи знайшли потрібні стовпці
    If groupColumn = 0 Or codeColumn = 0 Then
        InfoLabel.Caption = "ПОМИЛКА: Не знайдено стовпці 'Група' або 'ЄДРПОУ'!"
        InfoLabel.ForeColor = RGB(255, 0, 0)  ' Червоний колір
        sourceWorkbook.Close SaveChanges:=False
        Exit Sub
    End If
    
    ' Використовуємо Dictionary для унікальних груп
    Set groups = CreateObject("Scripting.Dictionary")
    
    ' Зчитуємо унікальні групи
    For i = headerRow + 1 To lastRow
        groupName = sourceSheet.Cells(i, groupColumn).Value
        If groupName <> "" And Not IsNull(groupName) Then
            groups(groupName) = 1
        End If
    Next i
    
    ' Заповнюємо ComboBox
    cmbGroup.Clear
    For Each groupName In groups.Keys
        cmbGroup.AddItem groupName
    Next groupName
    
    ' Зберігаємо всі дані для подальшого використання
    ' Зчитуємо тільки потрібні стовпці
    ReDim groupData(1 To lastRow - headerRow, 1 To 2)
    For i = headerRow + 1 To lastRow
        groupData(i - headerRow, 1) = sourceSheet.Cells(i, codeColumn).Value  ' ЄДРПОУ
        groupData(i - headerRow, 2) = sourceSheet.Cells(i, groupColumn).Value ' Група
    Next i
    
    ' Закриваємо файл-джерело
    sourceWorkbook.Close SaveChanges:=False
    
    If cmbGroup.ListCount > 0 Then
        InfoLabel.Caption = "Успішно завантажено " & cmbGroup.ListCount & " груп. Виберіть групу."
        InfoLabel.ForeColor = RGB(0, 128, 0)  ' Зелений колір
    Else
        InfoLabel.Caption = "Увага: Групи не знайдено у файлі!"
        InfoLabel.ForeColor = RGB(255, 165, 0)  ' Помаранчевий колір
    End If
    
    Exit Sub
ErrorHandler:
    InfoLabel.Caption = "ПОМИЛКА: " & Err.Description
    InfoLabel.ForeColor = RGB(255, 0, 0)  ' Червоний колір
    If Not sourceWorkbook Is Nothing Then sourceWorkbook.Close SaveChanges:=False
End Sub

' Подія при виборі групи
Private Sub cmbGroup_Change()
    Dim i As Long
    Dim selectedGroup As String
    
    If cmbGroup.ListIndex = -1 Then Exit Sub
    selectedGroup = cmbGroup.Value
    ' Очищаємо ListBox
    lstCodes.Clear
    ' Якщо даних немає, виходимо
    If IsEmpty(groupData) Then Exit Sub
    ' Проходимо по всіх рядках даних
    For i = 1 To UBound(groupData, 1)
        If groupData(i, 2) = selectedGroup Then
            lstCodes.AddItem groupData(i, 1)
        End If
    Next i
    ' Виводимо інформацію в Label
    If lstCodes.ListCount > 0 Then
        InfoLabel.Caption = "Знайдено " & lstCodes.ListCount & " кодів для групи '" & selectedGroup & "'"
        ActiveSheet.Range("CurentGroup").Value = selectedGroup
        InfoLabel.ForeColor = RGB(0, 128, 0)  ' Зелений колір
    Else
        InfoLabel.Caption = "Кодів для групи '" & selectedGroup & "' не знайдено!"
        InfoLabel.ForeColor = RGB(255, 165, 0)  ' Помаранчевий колір
    End If
End Sub

' Перевірка існування таблиці в книзі
Function TableExists(tableName As String) As Boolean
    Dim ws As Worksheet
    Dim tbl As ListObject
    
    TableExists = False
    For Each ws In ThisWorkbook.Worksheets
        For Each tbl In ws.ListObjects
            If tbl.Name = tableName Then
                TableExists = True
                Exit Function
            End If
        Next tbl
    Next ws
End Function

Sub RefreshConnections(ByVal connectionName As String, ByVal isMessage As Boolean)
    Dim conn As WorkbookConnection
    Dim foundConnection As Boolean
    
    If isMessage Then
        Set frmUpdate = New UpdateForm
        frmUpdate.Show vbModeless
        DoEvents
        Application.ScreenUpdating = False
        Application.DisplayAlerts = False
    End If
    
    Application.StatusBar = "Іде процес оновлення..."
    On Error GoTo ErrorHandler
    
    foundConnection = False
    
    For Each conn In ThisWorkbook.Connections
        ' Перевіряємо скасування
        If isMessage And Not frmUpdate Is Nothing Then
            DoEvents
            If frmUpdate.Cancelled Then GoTo CancelledHandler
        End If
        
        If Len(conn.Name) >= Len(connectionName) Then
            If Right(conn.Name, Len(connectionName)) = connectionName Then
                conn.Refresh
                foundConnection = True
                Exit For
            End If
        End If
    Next conn
    
    If Not foundConnection Then
        MsgBox "Підключення '" & connectionName & "' не знайдено!", vbExclamation
    Else
        If isMessage Then
            If Not frmUpdate Is Nothing Then
                Unload frmUpdate
                Set frmUpdate = Nothing
            End If
            Application.ScreenUpdating = True
            Application.DisplayAlerts = True
            MsgBox "Оновлення завершено!", vbInformation
        End If
    End If
    
    GoTo CleanUp
    
CancelledHandler:
    If isMessage Then
        If Not frmUpdate Is Nothing Then
            Unload frmUpdate
            Set frmUpdate = Nothing
        End If
        Application.ScreenUpdating = True
        Application.DisplayAlerts = True
        MsgBox "Оновлення скасовано користувачем!", vbExclamation
    End If
    
CleanUp:
    Application.StatusBar = False
    Exit Sub
    
ErrorHandler:
    If Not frmUpdate Is Nothing Then
        Unload frmUpdate
        Set frmUpdate = Nothing
    End If
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    MsgBox "Помилка: " & Err.Description, vbCritical
    Application.StatusBar = False
End Sub

Sub ClearTableByName(ByVal tableName As String)
    Dim ws As Worksheet
    Dim tbl As ListObject
    Dim found As Boolean
    
    ' Перебираємо всі аркуші в пошуках таблиці
    For Each ws In ThisWorkbook.Worksheets
        On Error Resume Next
        Set tbl = ws.ListObjects(tableName)
        On Error GoTo 0
        
        If Not tbl Is Nothing Then
            found = True
            ' Перевіряємо, чи є в таблиці рядки даних
            If Not tbl.DataBodyRange Is Nothing Then
                ' Очищуємо вміст комірок, але залишаємо саму таблицю та її зв'язок з запитом
                tbl.DataBodyRange.Delete
                RefreshCustomerData
                'ClearContents
                MsgBox "Таблицю '" & tableName & "' на аркуші '" & ws.Name & "' успішно очищено."
            Else
                RefreshCustomerData
                MsgBox "Таблиця '" & tableName & "' на аркуші '" & ws.Name & "' вже порожня."
            End If
            ws.Range("CurentGroup").Value = ""
            Exit For ' Припускаємо, що назви таблиць у книзі унікальні
        End If
    Next ws
    
    If Not found Then
        MsgBox "Таблицю з ім'ям '" & tableName & "' не знайдено в жодному аркуші книги.", vbExclamation
    End If
End Sub

Private Sub LoadGroups()
    Dim sourceWorkbook As Workbook
    Dim sourceSheet As Worksheet
    Dim lastRow As Long
    Dim lastCol As Long
    Dim i As Long, j As Long
    Dim groups As Object
    Dim groupName As Variant
    Dim headerRow As Long
    
    ' Шлях до файлу на робочому столі
    Dim filePath As String
    filePath = "%USERPROFILE%\Desktop\counterpartGroups.xlsx"
    
    ' Оновлюємо статус
    InfoLabel.Caption = "Завантаження файлу counterpartGroups.xlsx..."
    InfoLabel.ForeColor = RGB(0, 0, 255)  ' Синій колір
    DoEvents
    
    ' Перевіряємо чи існує файл
    If Dir(filePath) = "" Then
        InfoLabel.Caption = "ПОМИЛКА: Файл не знайдено: " & filePath
        InfoLabel.ForeColor = RGB(255, 0, 0)  ' Червоний колір
        Exit Sub
    End If
    
    ' Відкриваємо файл-джерело
    On Error GoTo ErrorHandler
    Set sourceWorkbook = Workbooks.Open(filePath, ReadOnly:=True)
    Set sourceSheet = sourceWorkbook.Worksheets(1)
    
    ' Визначаємо останній заповнений рядок і стовпець
    lastRow = sourceSheet.Cells(sourceSheet.Rows.Count, 1).End(xlUp).Row
    lastCol = sourceSheet.Cells(1, sourceSheet.Columns.Count).End(xlToLeft).Column
    
    ' Шукаємо заголовки стовпців (у першому рядку)
    headerRow = 1
    groupColumn = 0
    codeColumn = 0
    
    For j = 1 To lastCol
        If sourceSheet.Cells(headerRow, j).Value = "Група" Then
            groupColumn = j
        ElseIf sourceSheet.Cells(headerRow, j).Value = "ЄДРПОУ" Then
            codeColumn = j
        End If
    Next j
    
    ' Перевіряємо, чи знайшли потрібні стовпці
    If groupColumn = 0 Or codeColumn = 0 Then
        InfoLabel.Caption = "ПОМИЛКА: Не знайдено стовпці 'Група' або 'ЄДРПОУ'!"
        InfoLabel.ForeColor = RGB(255, 0, 0)  ' Червоний колір
        sourceWorkbook.Close SaveChanges:=False
        Exit Sub
    End If
    
    ' Використовуємо Dictionary для унікальних груп
    Set groups = CreateObject("Scripting.Dictionary")
    
    ' Зчитуємо унікальні групи
    For i = headerRow + 1 To lastRow
        groupName = sourceSheet.Cells(i, groupColumn).Value
        If groupName <> "" And Not IsNull(groupName) Then
            groups(groupName) = 1
        End If
    Next i
    
    ' Заповнюємо ComboBox
    cmbGroup.Clear
    For Each groupName In groups.Keys
        cmbGroup.AddItem groupName
    Next groupName
    
    ' Зберігаємо всі дані для подальшого використання
    ' Зчитуємо тільки потрібні стовпці
    ReDim groupData(1 To lastRow - headerRow, 1 To 2)
    For i = headerRow + 1 To lastRow
        groupData(i - headerRow, 1) = sourceSheet.Cells(i, codeColumn).Value  ' ЄДРПОУ
        groupData(i - headerRow, 2) = sourceSheet.Cells(i, groupColumn).Value ' Група
    Next i
    
    ' Закриваємо файл-джерело
    sourceWorkbook.Close SaveChanges:=False
    
    If cmbGroup.ListCount > 0 Then
        InfoLabel.Caption = "Успішно завантажено " & cmbGroup.ListCount & " груп. Виберіть групу."
        InfoLabel.ForeColor = RGB(0, 128, 0)  ' Зелений колір
    Else
        InfoLabel.Caption = "Увага: Групи не знайдено у файлі!"
        InfoLabel.ForeColor = RGB(255, 165, 0)  ' Помаранчевий колір
    End If
    
    Exit Sub
ErrorHandler:
    InfoLabel.Caption = "ПОМИЛКА: " & Err.Description
    InfoLabel.ForeColor = RGB(255, 0, 0)  ' Червоний колір
    If Not sourceWorkbook Is Nothing Then sourceWorkbook.Close SaveChanges:=False
End Sub

Sub RefreshConnections(ByVal connectionName As String, ByVal isMessage As Boolean)
    Dim conn As WorkbookConnection
    Dim foundConnection As Boolean
    
    If isMessage Then
        Set frmUpdate = New UpdateForm
        frmUpdate.Show vbModeless
        DoEvents
        Application.ScreenUpdating = False
        Application.DisplayAlerts = False
    End If
    
    Application.StatusBar = "Іде процес оновлення..."
    On Error GoTo ErrorHandler
    
    foundConnection = False
    
    For Each conn In ThisWorkbook.Connections
        ' Перевіряємо скасування
        If isMessage And Not frmUpdate Is Nothing Then
            DoEvents
            If frmUpdate.Cancelled Then GoTo CancelledHandler
        End If
        
        If Len(conn.Name) >= Len(connectionName) Then
            If Right(conn.Name, Len(connectionName)) = connectionName Then
                conn.Refresh
                foundConnection = True
                Exit For
            End If
        End If
    Next conn
    
    If Not foundConnection Then
        MsgBox "Підключення '" & connectionName & "' не знайдено!", vbExclamation
    Else
        If isMessage Then
            If Not frmUpdate Is Nothing Then
                Unload frmUpdate
                Set frmUpdate = Nothing
            End If
            Application.ScreenUpdating = True
            Application.DisplayAlerts = True
            MsgBox "Оновлення завершено!", vbInformation
        End If
    End If
    
    GoTo CleanUp
    
CancelledHandler:
    If isMessage Then
        If Not frmUpdate Is Nothing Then
            Unload frmUpdate
            Set frmUpdate = Nothing
        End If
        Application.ScreenUpdating = True
        Application.DisplayAlerts = True
        MsgBox "Оновлення скасовано користувачем!", vbExclamation
    End If
    
CleanUp:
    Application.StatusBar = False
    Exit Sub
    
ErrorHandler:
    If Not frmUpdate Is Nothing Then
        Unload frmUpdate
        Set frmUpdate = Nothing
    End If
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    MsgBox "Помилка: " & Err.Description, vbCritical
    Application.StatusBar = False
End Sub

Sub ClearTableByName(ByVal tableName As String)
    Dim ws As Worksheet
    Dim tbl As ListObject
    Dim found As Boolean
    
    ' Перебираємо всі аркуші в пошуках таблиці
    For Each ws In ThisWorkbook.Worksheets

        On Error GoTo 0
        
        If Not tbl Is Nothing Then

            ' Перевіряємо, чи є в таблиці рядки даних
