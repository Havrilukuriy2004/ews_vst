Sub AddNewSheet(ByVal pSheetName As String, ByRef newSheet As Worksheet)

    Dim ws As Worksheet
    Dim sheetExists As Boolean
    Dim newSheetName As String
    Dim cnt As Integer

    sheetExists = True
    cnt = 0
    newSheetName = pSheetName
    Do While sheetExists
        For Each ws In ThisWorkbook.Sheets
            If ws.Name = newSheetName Then
                sheetExists = True
                Exit For
            End If
            sheetExists = False
        Next ws
        
        If sheetExists Then
            cnt = cnt + 1
            newSheetName = pSheetName + "_" + Trim(Str(cnt))
        End If
    Loop
    
    Set newSheet = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
    newSheet.Name = newSheetName
    'ws.Cells.ClearContents
    'ws.Cells.ClearFormats
    
End Sub

Sub GetDataBtnClick()

    Dim cn As Object
    Dim cmd As Object
    Dim rs As Object
    Dim ws As Worksheet
    Dim i As Long
    Dim j As Long

    If FilterForm.cbReportForm.ListIndex = -1 Then
        MsgBox "Треба вибрати тип форми", vbExclamation
        Exit Sub
    End If

        MsgBox "Треба вибрати рік", vbExclamation
        Exit Sub
    End If
    
    If FilterForm.cbQr.ListIndex = -1 Then
        MsgBox "Треба вибрати квартал", vbExclamation

    End If

                    .Shadow = False

                    .TintAndShade = 0
                    .ThemeFont = xlThemeFontMinor

                    .PatternColorIndex = xlAutomatic

Sub DeleteAllSheetsExceptMenu()

    Dim ws As Worksheet
    Dim sheetToDelete As Worksheet
    Dim confirmMsg As String

    Application.DisplayAlerts = False

    On Error Resume Next
    Set ws = ThisWorkbook.Sheets("Main")
    On Error GoTo 0

    If ws Is Nothing Then
        MsgBox "Аркуш з назвою 'Main' не знайдно, операцію скасовано.", vbExclamation
        Application.DisplayAlerts = True
        Exit Sub
    End If

    For Each ws In ThisWorkbook.Sheets
        If ws.Name <> "Main" Then
            Set sheetToDelete = ws
            sheetToDelete.Delete
        End If
    Next ws

    Application.DisplayAlerts = True

End Sub

            GetDataTypeName = "Integer"
        Case adBigInt

        Case adUnsignedTinyInt
            GetDataTypeName = "UnsignedTinyInt"
        Case adUnsignedSmallInt
            GetDataTypeName = "UnsignedSmallInt"
        Case adUnsignedInt
            GetDataTypeName = "UnsignedInt"
        Case adUnsignedBigInt
            GetDataTypeName = "UnsignedBigInt"
        Case adSingle
            GetDataTypeName = "Single"
        Case adDouble
            GetDataTypeName = "Double"
        Case adCurrency
            GetDataTypeName = "Currency"
        Case adDecimal
            GetDataTypeName = "Decimal"
        Case adNumeric
            GetDataTypeName = "Numeric"
        Case adBoolean
            GetDataTypeName = "Boolean"
        Case adError
            GetDataTypeName = "Error"
        Case adUserDefined
            GetDataTypeName = "UserDefined"
        Case adVariant
            GetDataTypeName = "Variant"
        Case adGUID
            GetDataTypeName = "GUID"
        Case adDate
            GetDataTypeName = "Date"
        Case adDBDate
            GetDataTypeName = "DBDate"
        Case adDBTime
            GetDataTypeName = "DBTime"
        Case adDBTimeStamp
            GetDataTypeName = "DBTimeStamp"
        Case adChar
            GetDataTypeName = "Char"
        Case adWChar
            GetDataTypeName = "WChar"
        Case adBSTR
            GetDataTypeName = "BSTR"
        Case adLongVarChar
            GetDataTypeName = "LongVarChar"
        Case adLongVarWChar
            GetDataTypeName = "LongVarWChar"

Sub ExportFields()
    Dim cn As Object
    Dim cmd As Object
    Dim rs As Object
    Dim forms(7) As String
    Dim nameTabs(7) As String
    Dim ws As Worksheet
    Dim sheetExists As Boolean
    
    forms(0) = "Form1"
    forms(1) = "Form1m2m"
    forms(2) = "Form1ms2ms"
    forms(3) = "Form2"
    forms(4) = "Form3Direct"
    forms(5) = "Form3Undirect"
    forms(6) = "CollectReport"
    forms(7) = "CollectReport3"
    
    nameTabs(0) = "reportForm1"
    nameTabs(1) = "reportForm1m2m"
    nameTabs(2) = "reportForm1ms2ms"
    nameTabs(3) = "reportForm2"
    nameTabs(4) = "reportForm3Direct"
    nameTabs(5) = "reportForm3Undirect"
    nameTabs(6) = "v_CollectReport"
    nameTabs(7) = "v_CollectReport_3_33"
    
    sheetExists = False
    For Each s In ThisWorkbook.Sheets
        If s.Name = "Fields" Then
            sheetExists = True
            Set ws = ThisWorkbook.Sheets("Fields")
            Exit For
        End If
    Next s
        
    If Not sheetExists Then
        Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        ws.Name = "Fields"
    End If
    
    ws.Cells.ClearContents
    ws.Cells.ClearFormats
   
    ' --- Налаштування підключення до бази даних SQL Server ---
    Const DB_SERVER As String = "clientprofile-db-sqls.bank.lan" ' Ім'я SQL Server
    Const DB_NAME As String = "ClientProfile" ' Ім'я бази даних

    connString = "Provider=SQLOLEDB;" & _
                 "Data Source=" & DB_SERVER & ";" & _
                 "Initial Catalog=" & DB_NAME & ";" & _
                 "Integrated Security=SSPI;"

    On Error GoTo ErrorHandler

    Set cn = CreateObject("ADODB.Connection")
    cn.Open connString

    Dim f As Integer
   
    For f = 0 To 7
        Set cmd = CreateObject("ADODB.Command")
        Set cmd.ActiveConnection = cn
        cmd.CommandText = "SELECT TOP 1 * FROM finrep." & nameTabs(f)
        
        Set rs = cmd.Execute
        
        If Not rs Is Nothing Then ' Перевіряємо, чи об'єкт Recordset був успішно створений
            If rs.State = adStateOpen Then ' Перевіряємо, чи Recordset відкритий
                If rs.EOF And rs.BOF Then ' Перевіряємо, чи Recordset порожній (EOF - End Of File, BOF - Beginning Of File)
                    MsgBox "Запит не повернув жодних даних для форми " + nameTabs(f) + ".", vbInformation
                Else
    
                    ' --- Запис назв колонок ---
                    ws.Cells(1, (2 * f) + 1).Value = forms(f)
                    ws.Cells(1, (2 * f) + 2).Value = "Types " + forms(f)
                    
                    For i = 0 To rs.Fields.Count - 1
                        ws.Cells(i + 2, (2 * f) + 1).Value = "[" & rs.Fields(i).Name & "]"
                        ws.Cells(i + 2, (2 * f) + 2).Value = GetDataTypeName(rs.Fields(i).Type)
                    Next i
                
                End If
            Else
                MsgBox "Recordset не був відкритий після виконання збереженої процедури для форми " + forms(f) + ". Перевірте збережену процедуру та параметри.", vbExclamation
            End If
        Else
            MsgBox "Не вдалося отримати Recordset після виконання збереженої процедури для форми " + forms(f) + ". Можливо, процедура не повертає результат.", vbExclamation
        End If
            
        Set rs = Nothing
        Set cmd = Nothing

    Next

    ws.Activate
    Columns("A:P").Select
    With Selection.Font
        .Name = "Calibri"
        .Size = 9
    End With
    
    ws.Range(ws.Cells(1, 1), ws.Cells(1, (UBound(forms) + 1) * 2)).Select
    Selection.Font.Bold = True
    With Selection.Interior
        .Pattern = xlSolid
        .PatternColorIndex = xlAutomatic
        .ThemeColor = xlThemeColorAccent1
        .TintAndShade = 0.599993896298105
        .PatternTintAndShade = 0
    End With

    ws.UsedRange.Columns.AutoFit
    ws.Cells(1, 1).Select

Exit Sub

ErrorHandler:
    MsgBox "Виникла помилка: " & Err.Description, vbCritical
    If Not cn Is Nothing Then
        If cn.State = adStateOpen Then cn.Close
    End If
    If Not rs Is Nothing Then
        If rs.State = adStateOpen Then rs.Close
    End If
    
    Set cn = Nothing
    Set rs = Nothing
    Set cmd = Nothing

End Sub

Sub AddNewSheet(ByVal pSheetName As String, ByRef newSheet As Worksheet)

    Dim ws As Worksheet
    Dim sheetExists As Boolean
    Dim newSheetName As String
    Dim cnt As Integer

    sheetExists = True
    cnt = 0
    newSheetName = pSheetName
    Do While sheetExists
        For Each ws In ThisWorkbook.Sheets
            If ws.Name = newSheetName Then
                sheetExists = True
                Exit For
            End If
            sheetExists = False
        Next ws
        
        If sheetExists Then
            cnt = cnt + 1
            newSheetName = pSheetName + "_" + Trim(Str(cnt))
        End If
    Loop
    
    Set newSheet = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
    newSheet.Name = newSheetName
    'ws.Cells.ClearContents
    'ws.Cells.ClearFormats
    
End Sub

Sub GetDataBtnClick()

    Dim cn As Object
    Dim cmd As Object
    Dim rs As Object
    Dim ws As Worksheet
    Dim i As Long
    Dim j As Long

    If FilterForm.cbReportForm.ListIndex = -1 Then
        MsgBox "Треба вибрати тип форми", vbExclamation
        Exit Sub
    End If
    
    If FilterForm.cbYear.ListIndex = -1 Then
        MsgBox "Треба вибрати рік", vbExclamation
        Exit Sub
    End If
    
    If FilterForm.cbQr.ListIndex = -1 Then
        MsgBox "Треба вибрати квартал", vbExclamation
        Exit Sub
    End If
    
    If FilterForm.tbOKPO.Text = "" Then
        MsgBox "Треба вказати список ОКПО через кому    ", vbExclamation
        Exit Sub
    End If
    
    ' --- Налаштування підключення до бази даних SQL Server ---
    Const DB_SERVER As String = "clientprofile-db-sqls.bank.lan" ' Ім'я SQL Server
    Const DB_NAME As String = "ClientProfile" ' Ім'я бази даних

    connString = "Provider=SQLOLEDB;Data Source=" & DB_SERVER & ";Initial Catalog=" & DB_NAME & ";Integrated Security=SSPI;"

    On Error GoTo ErrorHandler

    Set cn = CreateObject("ADODB.Connection")
    cn.Open connString

    Set cmd = CreateObject("ADODB.Command")
    Set cmd.ActiveConnection = cn
    cmd.CommandText = "[finrep].[getReportFormInfo]"
    cmd.CommandType = adCmdStoredProc
    
    'EXEC [finrep].[getReportFormInfo] @reportForm = 2, @year = 2025, @qr = 1, @edrpou =  '42164147,40365742,43314719,24326663'
    
    cmd.Parameters.Append cmd.CreateParameter("@reportForm", adInteger, adParamInput, , FilterForm.cbReportForm.ListIndex + 1)
    cmd.Parameters.Append cmd.CreateParameter("@year", adInteger, adParamInput, , FilterForm.cbYear.Value)
    cmd.Parameters.Append cmd.CreateParameter("@qr", adInteger, adParamInput, , FilterForm.cbQr.ListIndex + 1)
    cmd.Parameters.Append cmd.CreateParameter("@edrpou", adVarWChar, adParamInput, -1, FilterForm.tbOKPO.Value)
    
    Set rs = cmd.Execute
    
    If Not rs Is Nothing Then ' Перевіряємо, чи об'єкт Recordset був успішно створений
        If rs.State = adStateOpen Then ' Перевіряємо, чи Recordset відкритий
            If rs.EOF And rs.BOF Then ' Перевіряємо, чи Recordset порожній (EOF - End Of File, BOF - Beginning Of File)
                MsgBox "Збережена процедура не повернула жодних даних.", vbInformation
            Else

                Call AddNewSheet(FilterForm.cbReportForm.Value, ws)
                '    ws.Cells.ClearContents
                ' ws.Cells(1, 1).Value = FilterForm.cbReportForm.Value + " за " + FilterForm.cbQr.Value + " " + FilterForm.cbYear.Value + " року."
                
                With ActiveSheet.Cells.Font
                    .Name = "Calibri"
                    .Size = 9
                    .Strikethrough = False
                    .Superscript = False
                    .Subscript = False
                    .OutlineFont = False
                    .Shadow = False
                    .Underline = xlUnderlineStyleNone
                    .ThemeColor = xlThemeColorLight1
                    .TintAndShade = 0
                    .ThemeFont = xlThemeFontMinor
                End With

                ' --- Запис назв колонок ---
                For i = 0 To rs.Fields.Count - 1
                    ws.Cells(1, i + 1).Value = rs.Fields(i).Name
                Next i
            
                ws.Range(ws.Cells(1, 1), ws.Cells(1, rs.Fields.Count)).Select
                Selection.Font.Bold = True
                With Selection.Interior
                    .Pattern = xlSolid
                    .PatternColorIndex = xlAutomatic
                    .ThemeColor = xlThemeColorAccent1
                    .TintAndShade = 0.599993896298105
                    .PatternTintAndShade = 0
                End With
                
                ' --- Запис даних ---
                ws.Range("A2").CopyFromRecordset rs
                
                Cells.EntireColumn.AutoFit
            
            ws.UsedRange.Columns.AutoFit
            ws.Cells(1, 1).Select
            
            'MsgBox "Дані успішно завантажено!", vbInformation
                
            End If
        Else
            MsgBox "Recordset не був відкритий після виконання збереженої процедури. Перевірте збережену процедуру та параметри.", vbExclamation
        End If
    Else
        MsgBox "Не вдалося отримати Recordset після виконання збереженої процедури. Можливо, процедура не повертає результат.", vbExclamation
    End If
    
Exit Sub

ErrorHandler:
    MsgBox "Виникла помилка: " & Err.Description, vbCritical
    If Not cn Is Nothing Then
        If cn.State = adStateOpen Then cn.Close
    End If
    If Not rs Is Nothing Then
        If rs.State = adStateOpen Then rs.Close
    End If
    
    Set rs = Nothing
    Set cmd = Nothing
    Set cn = Nothing

End Sub


Sub DeleteAllSheetsExceptMenu()

    Dim ws As Worksheet
    Dim sheetToDelete As Worksheet
    Dim confirmMsg As String

    Application.DisplayAlerts = False

    On Error Resume Next
    Set ws = ThisWorkbook."b'1\xa8\xa7'"
    On Error GoTo 0

    If ws Is Nothing Then
        MsgBox "Аркуш з назвою 'Main' не знайдно, операцію скасовано.", vbExclamation
        Application.DisplayAlerts = True
        Exit Sub
    End If

    For Each ws In ThisWorkbook.Sheets
        If ws.Name <> "Main" Then
            Set sheetToDelete = ws
            sheetToDelete.Delete
        End If
    Next ws

    Application.DisplayAlerts = True

End Sub

                    'Next

                Exit Sub
            End If
        End If
            
        Set rs = Nothing
        Set cmd = Nothing
    Next
    
    MsgBox "Завантаження завершено.", vbInformation

Exit Sub
    
ErrorHandler:
    MsgBox "Виникла помилка: " & Err.Description & " для форми " + forms(f) + ".", vbCritical

Sub ExportFields()
    Dim cn As Object
    Dim cmd As Object
    Dim rs As Object
    Dim forms(7) As String
    Dim nameTabs(7) As String
    Dim ws As Worksheet
    Dim sheetExists As Boolean
    
    forms(0) = "Form1"
    forms(1) = "Form1m2m"
    forms(2) = "Form1ms2ms"
    forms(3) = "Form2"
    forms(4) = "Form3Direct"
    forms(5) = "Form3Undirect"
    forms(6) = "CollectReport"
    forms(7) = "CollectReport3"
    
    nameTabs(0) = "reportForm1"
    nameTabs(1) = "reportForm1m2m"
    nameTabs(2) = "reportForm1ms2ms"
    nameTabs(3) = "reportForm2"
    nameTabs(4) = "reportForm3Direct"
    nameTabs(5) = "reportForm3Undirect"
    nameTabs(6) = "v_CollectReport"
    nameTabs(7) = "v_CollectReport_3_33"
    
    sheetExists = False
    For Each s In ThisWorkbook.Sheets
        If s.Name = "Fields" Then
            sheetExists = True
            Set ws = ThisWorkbook.Sheets("Fields")
            Exit For
        End If
    Next s
        
    If Not sheetExists Then
        Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        ws.Name = "Fields"
    End If
    
    ws.Cells.ClearContents
    ws.Cells.ClearFormats
   
    ' --- Налаштування підключення до бази даних SQL Server ---
    Const DB_SERVER As String = "clientprofile-db-sqls.bank.lan" ' Ім'я SQL Server
    Const DB_NAME As String = "ClientProfile" ' Ім'я бази даних

    connString = "Provider=SQLOLEDB;Data Source=" & DB_SERVER & ";Initial Catalog=" & DB_NAME & ";Integrated Security=SSPI;"

    On Error GoTo ErrorHandler

    Set cn = CreateObject("ADODB.Connection")
    cn.Open connString

    Dim f As Integer
   
    For f = 0 To 7
        Set cmd = CreateObject("ADODB.Command")
        Set cmd.ActiveConnection = cn
        cmd.CommandText = "SELECT TOP 1 * FROM finrep." & nameTabs(f)
        
        Set rs = cmd.Execute
        
        If Not rs Is Nothing Then ' Перевіряємо, чи об'єкт Recordset був успішно створений
            If rs.State = adStateOpen Then ' Перевіряємо, чи Recordset відкритий
                If rs.EOF And rs.BOF Then ' Перевіряємо, чи Recordset порожній (EOF - End Of File, BOF - Beginning Of File)
                    MsgBox "Запит не повернув жодних даних для форми " + nameTabs(f) + ".", vbInformation
                Else
    
                    ' --- Запис назв колонок ---
                    ws.Cells(1, (2 * f) + 1).Value = forms(f)
                    ws.Cells(1, (2 * f) + 2).Value = "Types " + forms(f)
                    
                    For i = 0 To rs.Fields.Count - 1
                        ws.Cells(i + 2, (2 * f) + 1).Value = "[" & rs.Fields(i).Name & "]"
                        ws.Cells(i + 2, (2 * f) + 2).Value = GetDataTypeName(rs.Fields(i).Type)
                    Next i
                
                End If
            Else
                MsgBox "Recordset не був відкритий після виконання збереженої процедури для форми " + forms(f) + ". Перевірте збережену процедуру та параметри.", vbExclamation
            End If
        Else
            MsgBox "Не вдалося отримати Recordset після виконання збереженої процедури для форми " + forms(f) + ". Можливо, процедура не повертає результат.", vbExclamation
        End If
            
        Set rs = Nothing
        Set cmd = Nothing

    Next

    ws.Activate
    Columns("A:P").Select
    With Selection.Font
        .Name = "Calibri"
        .Size = 9
    End With
    
    ws.Range(ws.Cells(1, 1), ws.Cells(1, (UBound(forms) + 1) * 2)).Select
    Selection.Font.Bold = True
    With Selection.Interior
        .Pattern = xlSolid
        .PatternColorIndex = xlAutomatic
        .ThemeColor = xlThemeColorAccent1
        .TintAndShade = 0.599993896298105
        .PatternTintAndShade = 0
    End With

    ws.UsedRange.Columns.AutoFit
    ws.Cells(1, 1).Select

Exit Sub

ErrorHandler:
    MsgBox "Виникла помилка: " & Err.Description, vbCritical
    If Not cn Is Nothing Then
        If cn.State = adStateOpen Then cn.Close
    End If
    If Not rs Is Nothing Then
        If rs.State = adStateOpen Then rs.Close
    End If
    
    Set cn = Nothing
    Set rs = Nothing
    Set cmd = Nothing

End Sub
